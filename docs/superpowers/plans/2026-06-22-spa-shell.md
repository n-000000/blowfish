# SPA Shell + Category Filtering Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Convert musictide to an app-shell SPA: category nav links filter the feed in-place with pushState; article cards swap into `#main-content` without full-page navigation; category term pages render as filtered home layouts.

**Architecture:** A new `#mt-feed-region` wrapper (`display:contents`) wraps the hero cell and `#mt-article-grid` inside `.mt-homepage-grid`. Category filtering swaps `#mt-feed-region`; article-swap swaps `#main-content`. A single `spa-router.js` handles both. Hugo pre-builds filtered term pages that are visually identical to the in-place filtered state.

**Tech Stack:** Hugo (blowfish fork), HTMX, vanilla JS, Tailwind (via `npm run build` in the blowfish fork)

- [ ] **Create `assets/js/spa-router.js`**

  ```javascript
  (function () {
    'use strict';

    var activeCategory = null; // current filter slug, null = all posts

    // --- URL helpers ---

    function categorySlugFromPath(path) {
      var m = (path || '').match(/^\/categories\/([^/]+)\//);
      return m ? m[1] : null;
    }

    function feedFragmentUrl(catSlug) {
      return catSlug
        ? '/categories/' + catSlug + '/feed-fragment.html'
        : '/posts/feed-fragment.html';
    }

    function articleFragmentUrl(path) {
      return path.replace(/\/?$/, '/') + 'fragment.html';
    }

    // --- Swap ---

    function swapFeed(catSlug, push) {
      htmx.ajax('GET', feedFragmentUrl(catSlug), {
        target: '#mt-feed-region',
        swap: 'innerHTML'
      });
      activeCategory = catSlug;
      applyNavActive(catSlug);
      if (push) {
        history.pushState(null, '', catSlug ? '/categories/' + catSlug + '/' : '/');
      }
    }

    function swapArticle(path, push) {
      htmx.ajax('GET', articleFragmentUrl(path), {
        target: '#main-content',
        swap: 'innerHTML'
      });
      if (push) {
        history.pushState(null, '', path);
      }
    }

    // --- Nav active state ---

    function applyNavActive(catSlug) {
      document.querySelectorAll('[data-category-link]').forEach(function (a) {
        var linkSlug = categorySlugFromPath(a.getAttribute('href') || '');
        if (linkSlug !== null && linkSlug === catSlug) {
          a.classList.add('mt-nav-active');
        } else {
          a.classList.remove('mt-nav-active');
        }
      });
    }

    // --- Wire nav links (idempotent via _spaWired flag) ---

    function wireNavLinks() {
      document.querySelectorAll('[data-category-link]').forEach(function (a) {
        if (a._spaWired) return;
        a._spaWired = true;
        a.addEventListener('click', function (e) {
          if (!document.getElementById('mt-feed-region')) return;
          e.preventDefault();
          var slug = categorySlugFromPath(a.getAttribute('href') || '');
          if (slug === activeCategory) {
            swapFeed(null, true);   // toggle off
          } else {
            swapFeed(slug, true);
          }
        });
      });
    }

    // --- Wire grid card links (idempotent) ---

    function wireGridCards() {
      document.querySelectorAll('.mt-grid-item a').forEach(function (a) {
        if (a._spaWired) return;
        a._spaWired = true;
        a.addEventListener('click', function (e) {
          var href = a.getAttribute('href') || '';
          if (!href || !href.startsWith('/') || href.startsWith('//')) return;
          e.preventDefault();
          swapArticle(href, true);
        });
      });
    }

    // --- Back / forward ---

    window.addEventListener('popstate', function () {
      var path = location.pathname;
      var catSlug = categorySlugFromPath(path);
      if (path === '/' || path === '') {
        swapFeed(null, false);
      } else if (catSlug) {
        swapFeed(catSlug, false);
      } else if (document.getElementById('main-content')) {
        swapArticle(path, false);
      }
    });

    // --- Post-swap: re-wire + update title ---

    document.addEventListener('htmx:afterSettle', function (e) {
      wireNavLinks();
      wireGridCards();
      if (e.detail && e.detail.target && e.detail.target.id === 'main-content') {
        var h1 = document.querySelector('#main-content h1');
        if (h1) {
          document.title = h1.textContent.trim();
        }
        window.scrollTo({ top: 0, behavior: 'instant' });
      }
    });

    // --- Init ---

    function init() {
      activeCategory = categorySlugFromPath(location.pathname);
      applyNavActive(activeCategory);
      wireNavLinks();
      wireGridCards();
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init);
    } else {
      init();
    }
  })();
  ```
