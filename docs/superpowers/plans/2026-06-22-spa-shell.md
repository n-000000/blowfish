# SPA Shell + Category Filtering — Implementation Plan (rev. 2026-06-22, keep-feed-alive)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
>
> **Run builds and verification commands inline (blocking), NOT in background. Wait for them to complete before committing or reporting done.**

**Goal:** Convert musictide to an app-shell SPA using the **keep-the-feed-alive** model: `#main-content` (on feed pages) holds two sibling regions — `#mt-feed-view` (the current feed) and `#mt-article-view` (an opened article). Category nav clicks re-render the feed region in place; clicking a card opens the article in the article region while the feed is *hidden, not destroyed*; Back reveals the feed exactly as left.

**Spec:** `docs/superpowers/specs/2026-06-21-spa-shell-design.md` (rev. 2026-06-22). **Read its Revision note + Architecture + Back/forward first.** This plan supersedes the obsolete two-target version of this file.

**Tech Stack:** Hugo (blowfish fork), HTMX 2.0.4 (loaded in musictide `extend-head.html`), vanilla JS, Tailwind v4 (`npm run build` in the fork).

---

## Architecture invariants (do not violate)

- **One swap target per concern.** View navigation toggles/replaces `#mt-feed-view` and `#mt-article-view`; infinite scroll stays internal to `#mt-article-grid`. Nothing targets `#main-content` directly anymore.
- **Opening an article never destroys the feed.** `#mt-feed-view` is hidden (`hidden` attr + `inert`), never emptied, so its DOM/scroll/sentinel survive for an exact Back.
- **Direct-load ↔ in-app parity via one shared partial.** `index.html` (home) and the home feed-fragment both render `home/hero.html`; `categories/term.html` and the category feed-fragment share the same structure. Never divergent copies.
- **Body layout class is toggled surgically** — flip only `mt-home` ↔ `mt-article-page`, never replace the whole `class` attribute.
- **A directly-loaded `/posts/{slug}/` is a standalone leaf** (`single.html`, no regions). The router degrades to hard navigation when `#mt-feed-view`/`#mt-article-view` are absent.

## Task 8: `spa-router.js` (keep-feed-alive) + partial + baseof include + extend-head cleanup

**Files:** `assets/js/spa-router.js`, `partials/spa-router.html`, `baseof.html`, **musictide** `extend-head.html`

- [ ] **Create `assets/js/spa-router.js`** — reference implementation (adapt as needed; it must satisfy the spec's *Filter behavior*, *Article-swap*, *Back/forward*, *Render context*):

  ```javascript
  (function () {
    'use strict';
    history.scrollRestoration = 'manual';

    function feedView()    { return document.getElementById('mt-feed-view'); }
    function articleView() { return document.getElementById('mt-article-view'); }

    var loadedFeedUrl = null;  // '/' or '/categories/{slug}/' currently in #mt-feed-view
    var feedScrollY   = 0;

    function catSlug(path)  { var m=(path||'').match(/^\/categories\/([^/]+)\/?$/); return m?m[1]:null; }
    function isArticle(p)   { return /^\/posts\/[^/]+\/?$/.test(p); }
    function feedUrlFor(p)  { var s=catSlug(p); return s?'/categories/'+s+'/':(p==='/'||p===''?'/':null); }
    function feedFragUrl(u) { return u==='/' ? '/posts/feed-fragment.html' : u+'feed-fragment.html'; }
    function artFragUrl(p)  { return p.replace(/\/?$/,'/')+'fragment.html'; }

    function setLayout(w){ var b=document.body.classList; b.toggle('mt-home',w==='home'); b.toggle('mt-article-page',w==='article'); }
    function closeMenu(){ var t=document.getElementById('mobile-menu-toggle'); if(t)t.checked=false; }

    function showFeed(){ var f=feedView(),a=articleView(); if(a){a.hidden=true;a.setAttribute('inert','');a.innerHTML='';} if(f){f.hidden=false;f.removeAttribute('inert');} setLayout('home'); }
    function showArticle(){ var f=feedView(),a=articleView(); if(f){f.hidden=true;f.setAttribute('inert','');} if(a){a.hidden=false;a.removeAttribute('inert');} setLayout('article'); }

    function applyNavActive(feedUrl){ var act=catSlug(feedUrl);
      document.querySelectorAll('[data-category-link]').forEach(function(l){
        var s=catSlug((l.getAttribute('href')||'').replace(/\/?$/,'/'));
        l.classList.toggle('mt-nav-active', s!==null && s===act);
      });
    }

    function renderFeed(feedUrl, push){ var f=feedView();
      if(!f){ location.href=feedUrl; return; }
      htmx.ajax('GET', feedFragUrl(feedUrl), {target:'#mt-feed-view', swap:'innerHTML'});
      loadedFeedUrl=feedUrl; showFeed(); applyNavActive(feedUrl); closeMenu(); window.scrollTo(0,0);
      if(push) history.pushState({view:'feed',feedUrl:feedUrl}, '', feedUrl);
    }

    function openArticle(path, push){ var a=articleView();
      if(!a){ location.href=path; return; }
      feedScrollY=window.scrollY;
      if(push) history.replaceState({view:'feed',feedUrl:loadedFeedUrl,scrollY:feedScrollY}, '', location.pathname);
      htmx.ajax('GET', artFragUrl(path), {target:'#mt-article-view', swap:'innerHTML'});
      showArticle(); closeMenu(); window.scrollTo(0,0);
      if(push) history.pushState({view:'article',path:path}, '', path);
    }

    window.addEventListener('popstate', function(e){
      var path=location.pathname;
      if(isArticle(path)){
        if(!articleView()||!articleView().innerHTML.trim()) openArticle(path,false); else showArticle();
        var h1=document.querySelector('#mt-article-view h1'); if(h1) document.title=h1.textContent.trim();
        window.scrollTo(0,0); return;
      }
      var feedUrl=feedUrlFor(path)||'/';
      if(feedUrl===loadedFeedUrl && feedView()){
        showFeed(); applyNavActive(feedUrl);
        window.scrollTo(0, (e.state&&e.state.scrollY)||feedScrollY||0);
      } else { renderFeed(feedUrl,false); }
    });

    function wireNav(){ document.querySelectorAll('[data-category-link]').forEach(function(l){
      if(l._spa)return; l._spa=1;
      l.addEventListener('click', function(e){
        if(!feedView()&&!articleView()) return;          // not an app page → let it navigate
        e.preventDefault();
        var s=catSlug((l.getAttribute('href')||'').replace(/\/?$/,'/'));
        var target=(s && ('/categories/'+s+'/')!==loadedFeedUrl) ? '/categories/'+s+'/' : '/';
        renderFeed(target, true);
      });
    }); }

    function wireCards(){ document.querySelectorAll('.mt-grid-item a').forEach(function(l){
      if(l._spa)return; l._spa=1;
      l.addEventListener('click', function(e){
        var href=l.getAttribute('href')||'';
        if(!href.startsWith('/')||href.startsWith('//')||!isArticle(href)) return;
        e.preventDefault(); openArticle(href, true);
      });
    }); }

    document.addEventListener('htmx:afterSettle', function(){ wireNav(); wireCards(); });

    function init(){ var p=location.pathname; loadedFeedUrl=feedUrlFor(p);
      history.replaceState({view:isArticle(p)?'article':'feed', feedUrl:loadedFeedUrl, path:p}, '', p);
      if(!isArticle(p)) applyNavActive(loadedFeedUrl||'/');
      wireNav(); wireCards();
    }
    if(document.readyState==='loading') document.addEventListener('DOMContentLoaded', init); else init();
  })();
  ```
